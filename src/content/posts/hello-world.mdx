## Code Examples

Let's test our syntax highlighting with some TypeScript code:

```typescript
interface BlogPost {
  title: string;
  description: string;
  date: string;
  tags: string[];
  draft: boolean;
}

const createPost = (data: BlogPost): BlogPost => {
  return {
    ...data,
    date: new Date().toISOString(),
  };
};

// Example usage
const myPost = createPost({
  title: "My Awesome Post",
  description: "This is a great post about coding",
  date: "2024-01-15",
  tags: ["typescript", "programming"],
  draft: false,
});

console.log(myPost);
```

And here's some React JSX:

```jsx
import React from 'react';

const BlogCard = ({ title, description, date, tags }) => {
  return (
    <article className="blog-card">
      <h2>{title}</h2>
      <p className="description">{description}</p>
      <time>{new Date(date).toLocaleDateString()}</time>
      <div className="tags">
        {tags.map(tag => (
          <span key={tag} className="tag">#{tag}</span>
        ))}
      </div>
    </article>
  );
};

export default BlogCard;
```
